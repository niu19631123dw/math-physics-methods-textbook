\begin{tikzpicture}
\begin{axis}[
  width=10cm, height=4.6cm,
  xlabel={$x$}, ylabel={$y$},
  xmin=-1.6, xmax=1.6, ymin=0, ymax=2.6,
  legend style={font=\footnotesize, at={(0.98,0.95)}, anchor=north east},
  grid=major, grid style={black!8}
]
% 悬链线 y = a cosh(x/a)，取 a=1，最低点在 (0,1)
\addplot[very thick, blue!70, domain=-1.5:1.5, samples=120] {cosh(x)};
% 抛物线拟合 y = 1 + x^2/2（在最低点的小垂度近似）
\addplot[thick, red!70, dashed, domain=-1.5:1.5, samples=120] {1 + x^2/2};
\legend{悬链线 $a\cosh(x/a)$, 抛物线近似}
\end{axis}
\end{tikzpicture}
\caption{悬链线与抛物线近似。小垂度（$|x|\ll a$）时两者几乎重合——这就是工程上「用抛物线近似悬链线」的依据；垂度一大，蓝线与红虚线明显分开。}
\end{figure}
